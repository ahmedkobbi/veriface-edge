    42→interface PremiumSpinnerProps {

    48→export function PremiumSpinner({ size = 'md', variant = 'gradient', className }: PremiumSpinnerProps) {
    49→  const sizes = { xs: 12, sm: 16, md: 24, lg: 32, xl: 48 }
    50→  const px = sizes[size]
    51→
    52→  if (variant === 'dots') {
    53→    return (
    54→      <div className={cn('inline-flex items-center gap-1', className)} style={{ height: px }}>
    55→        {[0, 1, 2].map((i) => (
    56→          <div
    57→            key={i}
    58→            className="rounded-full bg-gradient-to-br from-emerald-400 to-cyan-400"
    59→            style={{
    60→              width: px / 3,
    61→              height: px / 3,
    62→              animation: `pulse-dot 1.4s ease-in-out ${i * 0.16}s infinite`,
    63→            }}
    64→          />
    65→        ))}
    66→        <style jsx>{`
    67→          @keyframes pulse-dot {
    68→            0%, 80%, 100% { transform: scale(0.6); opacity: 0.4; }
    69→            40% { transform: scale(1); opacity: 1; }
    70→          }
    71→        `}</style>
    72→      </div>
    73→    )
    74→  }
    75→
    76→  if (variant === 'bars') {
    77→    return (
    78→      <div className={cn('inline-flex items-end gap-0.5', className)} style={{ height: px }}>
    79→        {[0, 1, 2, 3].map((i) => (
    80→          <div
    81→            key={i}
    82→            className="w-1 rounded-full bg-gradient-to-t from-emerald-500 to-cyan-400"
    83→            style={{
    84→              height: '100%',
    85→              animation: `bar-bounce 1s ease-in-out ${i * 0.1}s infinite`,
    86→            }}
    87→          />
    88→        ))}
    89→        <style jsx>{`
    90→          @keyframes bar-bounce {
    91→            0%, 100% { transform: scaleY(0.3); }
    92→            50% { transform: scaleY(1); }
    93→          }
    94→        `}</style>
    95→      </div>
    96→    )
    97→  }
    98→
    99→  if (variant === 'pulse') {
   100→    return (
   101→      <div className={cn('relative', className)} style={{ width: px, height: px }}>
   102→        <div
   103→          className="absolute inset-0 rounded-full bg-emerald-500/30"

   111→          @keyframes pulse-ring {
   112→            0% { transform: scale(0.8); opacity: 0.8; }
   113→            100% { transform: scale(1.4); opacity: 0; }
   114→          }
   115→          @keyframes pulse-core {
   116→            0%, 100% { transform: scale(1); }
   117→            50% { transform: scale(0.9); }
   118→          }
   119→        `}</style>
   120→      </div>
   121→    )
   122→  }
   123→
   124→  // Default: gradient ring spinner
   125→  return (
   126→    <svg
   127→      className={cn('animate-spin', className)}
   128→      width={px}
   129→      height={px}
   130→      viewBox="0 0 24 24"
   131→      fill="none"
   132→    >
   133→      <defs>
   134→        <linearGradient id="spinner-gradient" x1="0%" y1="0%" x2="100%" y2="100%">
   135→          <stop offset="0%" stopColor="#10b981" />
   136→          <stop offset="50%" stopColor="#06b6d4" />
   137→          <stop offset="100%" stopColor="#3b82f6" />
   138→        </linearGradient>
   139→      </defs>
   140→      <circle
   141→        cx="12" cy="12" r="10"
   142→        stroke="rgba(255,255,255,0.08)"
   143→        strokeWidth="2"
   144→        fill="none"
   145→      />
   146→      <path
   147→        d="M12 2 A10 10 0 0 1 22 12"
   148→        stroke="url(#spinner-gradient)"
   149→        strokeWidth="2"
   150→        strokeLinecap="round"
   151→        fill="none"
   152→      />

   253→export function PremiumAlert({

   255→  title,

   265→    leave: { opacity: 0, transform: 'translateY(-8px) scale(0.98)' },
   266→    config: { tension: 300, friction: 20 },
   267→  })
   268→
   269→  const variants = {
   270→    info: {
   271→      border: 'border-cyan-500/20',

   306→        role="alert"

   308→          'relative flex items-start gap-3 rounded-xl border p-3 backdrop-blur-xl',

   317→          {title && (
   318→            <p className={cn('text-xs font-semibold mb-0.5', v.text)}>{title}</p>
   319→          )}
   320→          <div className="text-xs text-slate-300">{children}</div>
   321→        </div>
   322→        {dismissible && (
   323→          <button
   324→            onClick={handleDismiss}
   325→            className="flex-shrink-0 text-slate-500 hover:text-slate-300 transition-colors"
   326→            aria-label="Dismiss"
   327→          >
   328→            ✕
   329→          </button>
   330→        )}
   331→      </animated.div>
   332→    ),
   333→  )
   334→}

   336→// ===========================================================================
   337→// Premium Toast — glass + slide-in + auto-dismiss countdown
   338→// ===========================================================================
   339→
   340→export type ToastVariant = 'info' | 'success' | 'warning' | 'error'
   341→
   342→export interface PremiumToastData {
   343→  id: string
   344→  title: string
   345→  description?: string
   346→  variant?: ToastVariant
   347→  duration?: number  // ms, default 5000
   348→}
   349→
   350→interface PremiumToastProps {
   351→  toast: PremiumToastData
   352→  onDismiss: (id: string) => void
   353→}
   354→
   355→export function PremiumToast({ toast, onDismiss }: PremiumToastProps) {
   356→  const [progress, setProgress] = useState(100)
   357→  const duration = toast.duration ?? 5000
   358→
   359→  useEffect(() => {
   360→    const start = Date.now()
   361→    const interval = setInterval(() => {
   362→      const elapsed = Date.now() - start
   363→      const remaining = Math.max(0, 100 - (elapsed / duration) * 100)
   364→      setProgress(remaining)
   365→      if (remaining <= 0) {
   366→        clearInterval(interval)
   367→        onDismiss(toast.id)
   368→      }
   369→    }, 50)
   370→    return () => clearInterval(interval)
   371→  }, [duration, onDismiss, toast.id])
   372→
   373→  const variants = {
   374→    info: { border: 'border-cyan-500/20', accent: 'from-cyan-500 to-blue-500', icon: 'ℹ' },
   375→    success: { border: 'border-emerald-500/20', accent: 'from-emerald-500 to-cyan-500', icon: '✓' },
   376→    warning: { border: 'border-amber-500/20', accent: 'from-amber-500 to-orange-500', icon: '⚠' },
   377→    error: { border: 'border-red-500/20', accent: 'from-red-500 to-rose-500', icon: '✕' },
   378→  }
   379→  const v = variants[toast.variant ?? 'info']
   380→
   381→  return (
   382→    <animated.div
   383→      className={cn(
   384→        'relative w-full max-w-sm overflow-hidden rounded-xl border backdrop-blur-2xl',
   385→        'bg-slate-950/80 shadow-2xl shadow-black/40',
   386→        v.border,
   387→      )}
   388→    >
   389→      {/* Accent bar */}
   390→      <div className={cn('h-0.5 bg-gradient-to-r', v.accent)} />
   391→      <div className="flex items-start gap-3 p-3">
   392→        <div className={cn('flex-shrink-0 mt-0.5 text-sm font-bold bg-gradient-to-br bg-clip-text text-transparent', v.accent)}>
   393→          {v.icon}
   394→        </div>
   395→        <div className="flex-1 min-w-0">
   396→          <p className="text-xs font-semibold text-slate-100">{toast.title}</p>
   397→          {toast.description && (
   398→            <p className="text-[11px] text-slate-400 mt-0.5">{toast.description}</p>
   399→          )}
   400→        </div>
   401→        <button
   402→          onClick={() => onDismiss(toast.id)}
   403→          className="flex-shrink-0 text-slate-500 hover:text-slate-300 transition-colors text-xs"
   404→          aria-label="Dismiss"
   405→        >
   406→          ✕
   407→        </button>
   408→      </div>
   409→      {/* Countdown progress */}
   410→      <div className="h-0.5 bg-white/5">
   411→        <div
   412→          className={cn('h-full bg-gradient-to-r transition-all duration-75', v.accent)}
   413→          style={{ width: `${progress}%` }}
   414→        />
   415→      </div>
   416→    </animated.div>
   417→  )
   418→}

   456→interface PremiumTooltipProps {
   457→  content: ReactNode
   458→  children: ReactNode
   459→  side?: 'top' | 'bottom' | 'left' | 'right'
   460→  delay?: number  // ms
   461→}
   462→
   463→export function PremiumTooltip({ content, children, side = 'top', delay = 200 }: PremiumTooltipProps) {
   464→  const [visible, setVisible] = useState(false)
   465→  const timer = useRef<NodeJS.Timeout | null>(null)
   466→
   467→  const show = () => {
   468→    timer.current = setTimeout(() => setVisible(true), delay)
   469→  }
   470→  const hide = () => {
   471→    if (timer.current) clearTimeout(timer.current)
   472→    setVisible(false)
   473→  }
   474→
   475→  const positions = {
   476→    top: 'bottom-full left-1/2 -translate-x-1/2 mb-2',
   477→    bottom: 'top-full left-1/2 -translate-x-1/2 mt-2',
   478→    left: 'right-full top-1/2 -translate-y-1/2 mr-2',
   479→    right: 'left-full top-1/2 -translate-y-1/2 ml-2',
   480→  }
   481→
   482→  return (
   483→    <div className="relative inline-block" onMouseEnter={show} onMouseLeave={hide}>
   484→      {children}
   485→      {visible && (
   486→        <div
   487→          className={cn(
   488→            'absolute z-50 pointer-events-none',
   489→            'px-2.5 py-1.5 rounded-lg',
   490→            'backdrop-blur-xl bg-slate-950/80 border border-white/[0.08]',
   491→            'text-[11px] text-slate-200 whitespace-nowrap',
   492→            'shadow-xl shadow-black/30',
   493→            'animate-[tooltip-in_0.15s_ease-out]',
   494→            positions[side],
   495→          )}
   496→        >
   497→          {content}
   498→          <style jsx>{`
   499→            @keyframes tooltip-in {
   500→              from { opacity: 0; transform: scale(0.95); }
   501→              to { opacity: 1; transform: scale(1); }
   502→            }
   503→          `}</style>
   504→        </div>
   505→      )}
   506→    </div>
   507→  )
   508→}

   666→interface PremiumButtonProps extends ButtonHTMLAttributes<HTMLButtonElement> {
   667→  variant?: 'primary' | 'secondary' | 'ghost' | 'danger' | 'outline'
   668→  size?: 'xs' | 'sm' | 'md' | 'lg'
   669→  loading?: boolean
   670→  icon?: ReactNode
   671→  iconPosition?: 'left' | 'right'
   672→  magnetic?: boolean
   673→  magneticStrength?: number
   674→  children?: ReactNode
   675→}
   676→
   677→export function PremiumButton({
   678→  variant = 'primary',
   679→  size = 'md',
   680→  loading = false,
   681→  icon,
   682→  iconPosition = 'left',
   683→  magnetic = true,
   684→  magneticStrength = 0.25,
   685→  children,
   686→  className,
   687→  disabled,
   688→  ...props
   689→}: PremiumButtonProps) {
   690→  const ref = useRef<HTMLButtonElement>(null)
   691→  const [springs, api] = useSpring(() => ({
   692→    x: 0, y: 0, scale: 1,
   693→    config: { tension: 300, friction: 20 },
   694→  }))
   695→
   696→  const handleMouseMove = (e: MouseEvent<HTMLButtonElement>) => {
   697→    if (!magnetic || disabled || loading) return
   698→    const el = ref.current
   699→    if (!el) return
   700→    const rect = el.getBoundingClientRect()
   701→    const x = e.clientX - (rect.left + rect.width / 2)
   702→    const y = e.clientY - (rect.top + rect.height / 2)
   703→    api.start({ x: x * magneticStrength, y: y * magneticStrength, scale: 1.03 })
   704→  }
   705→
   706→  const handleMouseLeave = () => {
   707→    api.start({ x: 0, y: 0, scale: 1 })
   708→  }
   709→
   710→  const variants = {
   711→    primary: 'bg-gradient-to-r from-emerald-500 to-cyan-500 text-white shadow-lg shadow-emerald-500/25 hover:shadow-emerald-500/40',
   712→    secondary: 'backdrop-blur-xl bg-white/[0.06] text-slate-100 border border-white/[0.08] hover:bg-white/[0.1]',
   713→    ghost: 'bg-transparent text-slate-400 hover:bg-white/[0.04] hover:text-slate-200',
   714→    danger: 'bg-gradient-to-r from-red-500 to-rose-500 text-white shadow-lg shadow-red-500/25 hover:shadow-red-500/40',
   715→    outline: 'bg-transparent text-slate-200 border border-white/[0.12] hover:bg-white/[0.04] hover:border-white/[0.2]',
   716→  }
   717→
   718→  const sizes = {
   719→    xs: 'px-2 py-1 text-[10px] gap-1',
   720→    sm: 'px-3 py-1.5 text-xs gap-1.5',
   721→    md: 'px-4 py-2 text-sm gap-2',
   722→    lg: 'px-6 py-3 text-base gap-2.5',
   723→  }
   724→
   725→  return (
   726→    <animated.button
   727→      ref={ref}
   728→      onMouseMove={handleMouseMove}
   729→      onMouseLeave={handleMouseLeave}
   730→      disabled={disabled || loading}
   731→      style={springs}
   732→      className={cn(
   733→        'relative inline-flex items-center justify-center rounded-xl font-medium',
   734→        'transition-all duration-200',
   735→        'disabled:opacity-50 disabled:cursor-not-allowed',
   736→        'focus:outline-none focus:ring-2 focus:ring-emerald-500/30 focus:ring-offset-2 focus:ring-offset-slate-950',
   737→        variants[variant],
   738→        sizes[size],
   739→        className,
   740→      )}
   741→      {...props}
   742→    >
   743→      {loading && (
   744→        <span className="absolute inset-0 flex items-center justify-center">
   745→          <PremiumSpinner size="sm" variant="gradient" />
   746→        </span>
   747→      )}
   748→      <span className={cn('inline-flex items-center gap-inherit', loading && 'opacity-0')}>
   749→        {icon && iconPosition === 'left' && <span className="flex-shrink-0">{icon}</span>}
   750→        {children}
   751→        {icon && iconPosition === 'right' && <span className="flex-shrink-0">{icon}</span>}
   752→      </span>
   753→    </animated.button>
   754→  )
   755→}
